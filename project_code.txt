import os

project_dir = r"D:\PROJECTS\science-diagram-maker"  # change this
output_file = "project_code.txt"

# Only include meaningful code files
include_extensions = {
    ".py", ".js", ".ts", ".html", ".css",
    ".cpp", ".c", ".java", ".json", ".svg"
}

# Folders to ignore
exclude_dirs = {
    "node_modules", ".git", "__pycache__",
    "dist", "build", ".next", "venv", ".vscode"
}

# Files to ignore
exclude_files = {
    "package-lock.json", "yarn.lock"
}

# Max file size -> 200 KB
MAX_FILE_SIZE = 200 * 1024

total_lines = 0

with open(output_file, "w", encoding="utf-8") as outfile:
    for root, dirs, files in os.walk(project_dir):

        # Remove unwanted directories
        dirs[:] = [d for d in dirs if d not in exclude_dirs]

        for file in files:
            file_path = os.path.join(root, file)
            rel_path = os.path.relpath(file_path, project_dir)

            # Skip unwanted files
            if file in exclude_files:
                continue

            # Skip by extension
            _, ext = os.path.splitext(file)
            if ext.lower() not in include_extensions:
                continue

            # Skip minified files
            if ".min." in file:
                continue

            # Skip large files
            if os.path.getsize(file_path) > MAX_FILE_SIZE:
                continue

            try:
                with open(file_path, "r", encoding="utf-8") as infile:
                    lines = infile.readlines()

                # Skip files with too many lines (likely generated)
                if len(lines) > 2000:
                    continue

                outfile.write("\n" + "="*70 + "\n")
                outfile.write(f"FILE: {rel_path}\n")
                outfile.write("="*70 + "\n")

                outfile.writelines(lines)
                total_lines += len(lines)

            except:
                continue

print(f"✅ Done! Clean code exported. Total lines: {total_lines}")

        return {"results": json.loads(text)}
    except FileNotFoundError:

    except json.JSONDecodeError:

@app.post("/api/generate")
def generate(request: GenerateRequest):
    """Run the whole pipeline live for one concept."""
    global _last_finished

    if request.concept_id not in CONCEPTS:
        return error_response(404, f"Unknown concept id: {request.concept_id!r}")

    # Don't wait for the lock. If someone else is generating, say so immediately.
    if not _generate_lock.acquire(blocking=False):
        return error_response(
            429, "Another diagram is being generated right now. Please wait for it to finish."
        )
    try:
        if _last_finished is not None:
            wait = config.GENERATE_COOLDOWN_SECONDS - (time.monotonic() - _last_finished)
            if wait > 0:
                seconds = math.ceil(wait)
                return error_response(
                    429,
                    f"Please wait {seconds} more second(s) before generating again. "
                    "This protects the free API quota.",
                    headers={"Retry-After": str(seconds)},
                )
        try:
            return generate_diagram(request.concept_id, profile="live")
        except GenerationError as error:
            status = error.status_code if error.status_code in (429, 503) else 502
            return error_response(status, error.message)
        except Exception:
            log.exception("Unexpected error while generating")
            return error_response(
                500, "Something went wrong on the server while generating. Please try again."
            )
        finally:
            _last_finished = time.monotonic()
    finally:
        _generate_lock.release()

from app import config


def estimate_cost(input_tokens, output_tokens, thought_tokens=0, model=None):
    """Estimated paid-tier cost in USD for the selected model.

    Thinking tokens are billed like output tokens, so we add them to the output.
    The free tier actually charges $0; this is the paid-tier equivalent.
    """
    price = config.PRICES.get(model or config.MODEL_NAME, config.DEFAULT_PRICE)
    billed_output = output_tokens + thought_tokens
    cost = (
        (input_tokens / 1_000_000) * price["input"]
        + (billed_output / 1_000_000) * price["output"]
    )
    return round(cost, 6)

SYSTEM_PROMPT = f"""You are an expert science illustrator making animated diagrams

def get_client():
    """Create the Gemini client, with a friendly error if the key is missing."""
    if not config.GEMINI_API_KEY:
        raise RuntimeError(
            "GEMINI_API_KEY is missing. Put it in the .env file in the project root."
        )
    return genai.Client(api_key=config.GEMINI_API_KEY)


def build_prompt(concept):
    """Turn one concept dictionary into the user message."""
    return (
        f"Concept: {concept['name']} ({concept['subject']})\n\n"
        f"Explanation: {concept['description']}\n\n"
        f"The diagram must show: {concept['hint']}\n\n"
        "Return only the SVG."
    )



def build_correction_prompt(concept, previous_svg, failures):
    """The message for the validation retry: what went wrong, plus the previous answer."""
    problems = "\n".join(f"- {failure}" for failure in failures)
    return (
        build_prompt(concept)
        + "\n\nYour previous answer failed these automatic checks:\n"
        + problems
        + "\n\nHere is your previous answer:\n"
        + previous_svg[:30000]
        + "\n\nFix every problem listed above and return the complete corrected SVG. "
        "Return only the SVG, with no explanation."
    )



def extract_svg(text):
    """Cut out everything from '<svg' to the last '</svg>'.

    This removes markdown fences and any chatter the model added around the SVG.
    """
    start = text.find("<svg")
    end = text.rfind("</svg>")
    if start == -1 or end == -1 or end < start:
        raise ValueError("No complete <svg>...</svg> found in the model response")
    return text[start : end + len("</svg>")]


def call_gemini(prompt):
    """Send one prompt to Gemini. Return the raw text and the token usage."""
    client = get_client()
    response = client.models.generate_content(
        model=config.MODEL_NAME,
        contents=prompt,
        config=types.GenerateContentConfig(
            system_instruction=SYSTEM_PROMPT,
            max_output_tokens=config.MAX_OUTPUT_TOKENS,
        ),
    )

    usage = response.usage_metadata
    finish_reason = None
    if response.candidates:
        finish_reason = str(response.candidates[0].finish_reason)

    return {
        "text": response.text or "",
        "model": config.MODEL_NAME,
        # "or 0" protects us if a field comes back empty
        "input_tokens": (usage.prompt_token_count or 0) if usage else 0,
        "output_tokens": (usage.candidates_token_count or 0) if usage else 0,
        "thought_tokens": (usage.thoughts_token_count or 0) if usage else 0,
        "finish_reason": finish_reason,
    }


def generate_svg(concept):
    """Ask Gemini for an SVG for one concept."""
    reply = call_gemini(build_prompt(concept))
    reply["svg"] = extract_svg(reply["text"])
    return reply

RETRYABLE_CODES = {429, 503}


class GenerationError(Exception):
    """Raised when Gemini could not give us anything usable.

    .message is safe to show to a user on the web page.
    """

    def __init__(self, message, status_code=None, api_retries=0):
        super().__init__(message)
        self.message = message
        self.status_code = status_code
        self.api_retries = api_retries



# ------------------------------------------------------------ error helpers
def get_status_code(error):
    """Pull an HTTP-style code (429, 503...) out of an exception, or None."""
    for name in ("code", "status_code"):
        value = getattr(error, name, None)
        if isinstance(value, int):
            return value
    return None


def suggested_delay(error):
    """If Gemini says how long to wait ('retryDelay': '34s'), return seconds, else 0."""
    match = re.search(
        r"(?:retry\w*delay\W+|retry in\s+)(\d+(?:\.\d+)?)\s*s", str(error), re.I
    )
    return float(match.group(1)) if match else 0.0


def friendly_message(code, error):
    if code in (400, 401, 403) and "api key" in str(error).lower():
        return "Gemini rejected the API key. Check GEMINI_API_KEY."
    messages = {
        # ... the rest stays exactly as it was
        429: "Gemini's rate limit was reached. Free-tier limits are low, so please wait a minute and try again.",
        503: "Gemini is overloaded right now. Please try again shortly.",
        401: "Gemini rejected the API key. Check GEMINI_API_KEY.",
        403: "Gemini rejected the API key or this model is not allowed for it. Check GEMINI_API_KEY.",
        404: f"Model '{config.MODEL_NAME}' was not found. Check MODEL_NAME in config.py.",
        400: "Gemini rejected the request as invalid.",
    }
    if code in messages:
        return messages[code]
    return f"Gemini request failed ({type(error).__name__}): {str(error)[:200]}"


# --------------------------------------------------------------- API retry
def call_with_api_retry(call_model, prompt, profile, sleep):
    """Call the model. On 429/503 wait and retry, up to MAX_API_RETRIES times.

    Returns (reply, number_of_retries_used). Any other error stops immediately.
    """
    settings = config.RETRY_PROFILES[profile]
    delays = settings["delays"]
    retries = 0

    while True:
        try:
            return call_model(prompt), retries
        except Exception as error:
            code = get_status_code(error)
            retryable = code in RETRYABLE_CODES
            if not retryable or retries >= config.MAX_API_RETRIES:
                raise GenerationError(
                    friendly_message(code, error), code, api_retries=retries
                ) from error

            base = delays[min(retries, len(delays) - 1)]
            wait = min(max(base, suggested_delay(error)), settings["max_wait"])
            wait += random.uniform(0, 1)  # small random extra so retries don't all line up
            log.warning(
                "Gemini error %s (%s). Waiting %.1fs, then API retry %d of %d",
                code, str(error)[:150], wait, retries + 1, config.MAX_API_RETRIES,
            )
            sleep(wait)
            retries += 1


# ------------------------------------------------------------- one attempt
def _run_attempt(call_model, prompt, profile, sleep, kind):
    """One model call plus the five checks. Returns a dictionary describing it."""
    reply, api_retries = call_with_api_retry(call_model, prompt, profile, sleep)

    text = reply["text"]
    problem = None
    try:
        svg = gemini_client.extract_svg(text)
        svg_ok = True
    except ValueError as error:
        svg, svg_ok = text, False
        problem = str(error)
        if "MAX_TOKENS" in str(reply.get("finish_reason")):
            problem += " (the reply was cut off: MAX_TOKENS)"

    checks = validate_svg(svg)
    if problem:
        checks["valid_xml"] = {"passed": False, "reason": problem}

    return {
        "kind": kind,
        "svg": svg,            # the SVG, or the raw reply text if no SVG was found
        "svg_ok": svg_ok,
        "checks": checks,
        "error": None,
        "api_retries": api_retries,
        "model": reply.get("model"),
        "input_tokens": reply.get("input_tokens", 0),
        "output_tokens": reply.get("output_tokens", 0),
        "thought_tokens": reply.get("thought_tokens", 0),
        "finish_reason": reply.get("finish_reason"),
    }


def _failed_attempt(kind, error):
    """Record an attempt that died from an API error (no SVG, no tokens)."""
    return {
        "kind": kind, "svg": "", "svg_ok": False, "checks": None,
        "error": error.message, "api_retries": error.api_retries,
        "model": None, "input_tokens": 0, "output_tokens": 0,
        "thought_tokens": 0, "finish_reason": None,
    }


# ------------------------------------------------------------- main entry
def generate_diagram(concept_id, call_model=None, sleep=time.sleep, profile="batch"):
    """Generate, check and (if needed) correct one diagram. Returns a result dict.

    call_model: function(prompt) -> dict. Defaults to the real Gemini call.
    sleep:      function(seconds). Tests pass a fake so they don't actually wait.
    profile:    "batch" (patient) or "live" (for the web page).
    Raises GenerationError if the FIRST call fails completely.
    """
    if concept_id not in CONCEPTS:
        raise ValueError(f"Unknown concept id: {concept_id!r}")
    call_model = call_model or gemini_client.call_gemini

    concept = CONCEPTS[concept_id]
    prompt = gemini_client.build_prompt(concept)
    log.info("[%s] Generating: %s", concept_id, concept["name"])

    attempts = [_run_attempt(call_model, prompt, profile, sleep, "initial")]
    validation_retries = 0

    # The validation retry: only if checks failed, and only MAX_VALIDATION_RETRIES times
    while (
        not all_passed(attempts[-1]["checks"])
        and validation_retries < config.MAX_VALIDATION_RETRIES
    ):
        validation_retries += 1
        failures = failure_messages(attempts[-1]["checks"])
        log.info("[%s] Checks failed: %s", concept_id, " | ".join(failures))
        correction = gemini_client.build_correction_prompt(
            concept, attempts[-1]["svg"], failures
        )
        try:
            attempts.append(
                _run_attempt(call_model, correction, profile, sleep, "correction")
            )
        except GenerationError as error:
            # Keep attempt 1 instead of losing it. We save what we have, honestly.
            log.warning("[%s] Correction attempt failed: %s", concept_id, error.message)
            attempts.append(_failed_attempt("correction", error))
            break

    # Choose the final attempt: the last one that produced checks. If that one had no
    # SVG at all but an earlier one did, keep the earlier SVG.
    scored = [a for a in attempts if a["checks"] is not None]
    final = scored[-1]
    if not final["svg_ok"]:
        for earlier in reversed(scored):
            if earlier["svg_ok"]:
                final = earlier
                break

    # Totals across ALL attempts, including the failed first one
    total_input = sum(a["input_tokens"] for a in attempts)
    total_output = sum(a["output_tokens"] for a in attempts)
    total_thoughts = sum(a["thought_tokens"] for a in attempts)
    api_retries = sum(a["api_retries"] for a in attempts)
    passed = all_passed(final["checks"])
    used_model = final["model"] or scored[0]["model"] or config.MODEL_NAME
    price = config.PRICES.get(used_model, config.DEFAULT_PRICE)

    result = {
        "concept_id": concept_id,
        "concept_name": concept["name"],
        "subject": concept["subject"],
        "model": used_model,
        "svg": final["svg"] if final["svg_ok"] else "",
        "status": "passed" if passed else "failed",
        "checks": final["checks"],
        "tokens": {
            "input": total_input,
            "output": total_output,
            "thoughts": total_thoughts,
            "billed_output": total_output + total_thoughts,
        },
        "cost_usd": estimate_cost(total_input, total_output, total_thoughts, used_model),
        "pricing": {
            "input_per_million": price["input"],
            "output_per_million": price["output"],
            "note": "Estimated paid-tier equivalent. The free tier charges $0.",
        },
        "retries": {
            "api": api_retries,
            "validation": validation_retries,
            "total": api_retries + validation_retries,
        },
        "attempts": [
            {
                "attempt": number,
                "kind": a["kind"],
                "api_retries": a["api_retries"],
                "input_tokens": a["input_tokens"],
                "output_tokens": a["output_tokens"],
                "thought_tokens": a["thought_tokens"],
                "finish_reason": a["finish_reason"],
                "error": a["error"],
                "failures": failure_messages(a["checks"]) if a["checks"] else [],
            }
            for number, a in enumerate(attempts, start=1)
        ],
        "generated_at": datetime.now(timezone.utc).isoformat(timespec="seconds"),
    }

    log.info(
        "[%s] %s | tokens in=%d out=%d thoughts=%d | cost $%.6f | retries api=%d validation=%d",
        concept_id, result["status"].upper(), total_input, total_output,
        total_thoughts, result["cost_usd"], api_retries, validation_retries,
    )
    return result

"""Checks for AI-generated SVGs.

Every check returns (passed: bool, reason: str).
validate_svg() runs all five and returns a dictionary the web page can show.
"""
import re
import xml.etree.ElementTree as ET

from app import config

# A number like 12, -3.5, .5 or 1e3 (no capture groups, so findall returns whole numbers)
NUMBER_PATTERN = r"-?(?:\d+\.?\d*|\.\d+)(?:[eE][-+]?\d+)?"
_NUMBER = re.compile(NUMBER_PATTERN)
_TRANSLATE = re.compile(
    rf"^\s*translate\(\s*({NUMBER_PATTERN})\s*(?:[,\s]\s*({NUMBER_PATTERN}))?\s*\)\s*$"
)

ANIMATION_TAGS = {"animate", "animatetransform", "animatemotion", "set"}
# Children with these tags move their parent, so its static position means nothing
MOVING_TAGS = {"animatemotion", "animatetransform"}

# Parts of an SVG that are not drawn directly, so we don't measure them
SKIP_TAGS = {"defs", "clippath", "mask", "pattern", "symbol", "marker",
             "style", "title", "desc", "metadata"}

# Shapes we can't measure with simple maths. We count them but don't fail them.
UNCHECKED_TAGS = {"path", "image", "use"}

FORBIDDEN_TAGS = {"script", "foreignobject", "iframe", "embed", "object"}


# ----------------------------------------------------------------- helpers
def _local(tag):
    """'{http://www.w3.org/2000/svg}Circle' -> 'circle'"""
    return tag.split("}")[-1].lower()


def _num(elem, name, default=0.0):
    """Read a numeric attribute.

    Missing attribute -> default. Percentages or junk -> None ("can't measure").
    """
    raw = elem.get(name)
    if raw is None:
        return default
    raw = raw.strip()
    if "%" in raw:
        return None
    match = _NUMBER.match(raw)
    return float(match.group()) if match else None


def _first_number(raw, default=0.0):
    """First number in a string like '10 20 30' (text x can be a list)."""
    match = _NUMBER.search(raw or "")
    return float(match.group()) if match else default


def _text_content(elem):
    """All text inside an element, with whitespace tidied."""
    return " ".join("".join(elem.itertext()).split())


def parse_svg(svg_string):
    """Parse the SVG or raise an error. Used by check 1 and by validate_svg()."""
    lowered = svg_string.lower()
    # DOCTYPE/ENTITY tricks can make XML parsers misbehave. SVGs never need them.
    if "<!doctype" in lowered or "<!entity" in lowered:
        raise ValueError("SVG contains a DOCTYPE or ENTITY declaration, which is not allowed")
    root = ET.fromstring(svg_string)
    if _local(root.tag) != "svg":
        raise ValueError(f"Root element is <{_local(root.tag)}>, not <svg>")
    return root


# ----------------------------------------------------- check 1: valid XML
def check_valid_xml(svg_string):
    try:
        parse_svg(svg_string)
    except ET.ParseError as error:
        return False, f"Invalid XML: {error}"
    except ValueError as error:
        return False, str(error)
    return True, "Valid XML with an <svg> root element"


# --------------------------------------------------- check 2: inside viewBox
def _get_viewbox(root):
    """Return [min_x, min_y, width, height] or None if missing/invalid."""
    raw = root.get("viewBox")
    if not raw:
        return None
    try:
        numbers = [float(p) for p in re.split(r"[\s,]+", raw.strip())]
    except ValueError:
        return None
    if len(numbers) != 4 or numbers[2] <= 0 or numbers[3] <= 0:
        return None
    return numbers


def _css_rules(root):
    """Very small CSS reader: {'.title': 'font-size:24px;text-anchor:middle', 'text': ...}."""
    rules = {}
    for selectors, body in re.findall(r"([^{}@]+)\{([^{}]*)\}", _collect_css(root)):
        for selector in selectors.split(","):
            rules[selector.strip()] = rules.get(selector.strip(), "") + ";" + body
    return rules


def _effective_style(elem, rules):
    """Rules for 'text' and the element's classes, then its inline style (last wins)."""
    parts = [rules.get("text", "")]
    for cls in (elem.get("class") or "").split():
        parts.append(rules.get("." + cls, ""))
        parts.append(rules.get("text." + cls, ""))
    parts.append(elem.get("style", ""))
    return ";".join(parts)


def _text_box(elem, rules=None):
    """Estimate the box a <text> covers. Returns None if the text is empty."""
    content = _text_content(elem)
    if not content:
        return None
    style = _effective_style(elem, rules or {})

    # CSS (classes / inline style) beats the plain attribute, so look there first
    matches = re.findall(r"font-size\s*:\s*([\d.]+)", style)
    if matches:
        size = float(matches[-1])
    else:
        size = _num(elem, "font-size", None) or config.DEFAULT_FONT_SIZE

    matches = re.findall(r"text-anchor\s*:\s*(\w+)", style)
    anchor = matches[-1] if matches else elem.get("text-anchor", "start")

    x = _first_number(elem.get("x"))
    y = _first_number(elem.get("y"))
    width = config.CHAR_WIDTH_FACTOR * size * len(content)

    if anchor == "middle":
        x0 = x - width / 2
    elif anchor == "end":
        x0 = x - width
    else:
        x0 = x
    # Text sits ON the baseline y, so most of it is above y
    return (x0, y - size, x0 + width, y + 0.25 * size)


def _element_box(elem, tag, rules=None):
    """Return (x0, y0, x1, y1), the string 'unchecked', or None if not a shape."""
    if tag == "text":
        return _text_box(elem, rules)

    if tag == "circle":
        cx, cy, r = _num(elem, "cx"), _num(elem, "cy"), _num(elem, "r")
        if None in (cx, cy, r):
            return "unchecked"
        box = (cx - r, cy - r, cx + r, cy + r)
    elif tag == "ellipse":
        cx, cy = _num(elem, "cx"), _num(elem, "cy")
        rx, ry = _num(elem, "rx"), _num(elem, "ry")
        if None in (cx, cy, rx, ry):
            return "unchecked"
        box = (cx - rx, cy - ry, cx + rx, cy + ry)
    elif tag == "rect":
        x, y = _num(elem, "x"), _num(elem, "y")
        w, h = _num(elem, "width"), _num(elem, "height")
        if None in (x, y, w, h):
            return "unchecked"
        box = (x, y, x + w, y + h)
    elif tag == "line":
        x1, y1, x2, y2 = (_num(elem, n) for n in ("x1", "y1", "x2", "y2"))
        if None in (x1, y1, x2, y2):
            return "unchecked"
        box = (min(x1, x2), min(y1, y2), max(x1, x2), max(y1, y2))
    elif tag in ("polygon", "polyline"):
        numbers = [float(n) for n in _NUMBER.findall(elem.get("points", ""))]
        pairs = list(zip(numbers[0::2], numbers[1::2]))
        if not pairs:
            return "unchecked"
        xs, ys = [p[0] for p in pairs], [p[1] for p in pairs]
        box = (min(xs), min(ys), max(xs), max(ys))
    elif tag in UNCHECKED_TAGS:
        return "unchecked"
    else:
        return None  # a container like <g>, or something we ignore

    # A stroke is drawn half inside, half outside the shape's edge
    stroke = elem.get("stroke")
    if stroke and stroke != "none":
        half = (_num(elem, "stroke-width", 1.0) or 0.0) / 2
        box = (box[0] - half, box[1] - half, box[2] + half, box[3] + half)
    return box


def _collect_boxes(root):
    """Walk the whole SVG. Return (list of (tag, x0, y0, x1, y1), unchecked_count)."""
    boxes = []
    state = {"unchecked": 0}
    rules = _css_rules(root)

    def walk(elem, dx, dy):
        tag = _local(elem.tag)
        if tag in SKIP_TAGS:
            return

        # Support transform="translate(x, y)" only. Anything else -> skip this subtree.
        transform = elem.get("transform")
        if transform:
            match = _TRANSLATE.match(transform)
            if match:
                dx += float(match.group(1))
                dy += float(match.group(2) or 0)
            else:
                state["unchecked"] += 1
                return

        if any(_local(child.tag) in MOVING_TAGS for child in elem):
            state["unchecked"] += 1   # moved by animateMotion/animateTransform: skip it
            return

        result = _element_box(elem, tag, rules)
        if result == "unchecked":
            state["unchecked"] += 1
        elif result is not None:
            x0, y0, x1, y1 = result
            boxes.append((tag, x0 + dx, y0 + dy, x1 + dx, y1 + dy))

        for child in elem:
            walk(child, dx, dy)

    for child in root:
        walk(child, 0.0, 0.0)
    return boxes, state["unchecked"]


def check_viewbox_bounds(root):
    viewbox = _get_viewbox(root)
    if viewbox is None:
        return False, "Missing or invalid viewBox attribute"

    min_x, min_y, width, height = viewbox
    max_x, max_y = min_x + width, min_y + height
    tol = config.BOUNDS_TOLERANCE

    boxes, unchecked = _collect_boxes(root)
    problems = []
    for tag, x0, y0, x1, y1 in boxes:
        sides = []
        if x0 < min_x - tol:
            sides.append(f"left edge at {x0:.0f}")
        if y0 < min_y - tol:
            sides.append(f"top edge at {y0:.0f}")
        if x1 > max_x + tol:
            sides.append(f"right edge at {x1:.0f}")
        if y1 > max_y + tol:
            sides.append(f"bottom edge at {y1:.0f}")
        if sides:
            problems.append(f"<{tag}> {', '.join(sides)}")

    if problems:
        shown = "; ".join(problems[:3])
        more = f" (+{len(problems) - 3} more)" if len(problems) > 3 else ""
        return False, (f"{len(problems)} element(s) outside the "
                       f"{width:.0f}x{height:.0f} viewBox: {shown}{more}")

    note = f"; {unchecked} part(s) not checked (paths or complex transforms)" if unchecked else ""
    return True, f"All {len(boxes)} measured shapes are inside the viewBox{note}"


# ---------------------------------------------- check 3: title and labels
def check_title_and_labels(root):
    titles = [e for e in root.iter() if _local(e.tag) == "title" and _text_content(e)]
    labels = [e for e in root.iter() if _local(e.tag) == "text" and _text_content(e)]

    problems = []
    if not titles:
        problems.append("No non-empty <title> element")
    if len(labels) < config.MIN_LABELS:
        problems.append(f"Only {len(labels)} text label(s), need at least {config.MIN_LABELS}")

    if problems:
        return False, "; ".join(problems)
    return True, f"Title found and {len(labels)} text labels"


# ------------------------------------------------------ check 4: animation
def _collect_css(root):
    """All CSS in <style> tags and style="..." attributes, joined together."""
    parts = []
    for elem in root.iter():
        if _local(elem.tag) == "style":
            parts.append("".join(elem.itertext()))
        if elem.get("style"):
            parts.append(elem.get("style"))
    return "\n".join(parts)


def check_animation(root):
    smil = [e for e in root.iter() if _local(e.tag) in ANIMATION_TAGS]
    if smil:
        return True, f"Found {len(smil)} SMIL animation element(s)"

    css = _collect_css(root)
    names = re.findall(r"@keyframes\s+([\w-]+)", css)
    for name in names:
        # Is there an "animation: ..." or "animation-name: ..." declaration using this name?
        pattern = r"animation[\w-]*\s*:[^;}]*\b" + re.escape(name) + r"\b"
        if re.search(pattern, css):
            return True, f"CSS animation '{name}' is defined and used"

    if names:
        return False, "@keyframes defined but no animation declaration uses them"
    return False, "No animation found (no SMIL tags and no CSS @keyframes)"


# -------------------------------------------------- check 5: security (mine)
def check_security(root):
    problems = []

    for elem in root.iter():
        tag = _local(elem.tag)
        if tag in FORBIDDEN_TAGS:
            problems.append(f"forbidden element <{tag}>")

        for attr, value in elem.attrib.items():
            name = _local(attr)
            if name.startswith("on"):
                problems.append(f"event handler attribute '{name}'")
            if name == "href" and not value.strip().startswith("#"):
                problems.append(f"external link in href: {value[:40]}")
            if "javascript:" in value.lower():
                problems.append("javascript: URL in an attribute")
            for target in re.findall(r"url\(([^)]*)\)", value, re.I):
                if not target.strip(" '\"").startswith("#"):
                    problems.append(f"external url() in attribute '{name}'")

    css = _collect_css(root)
    if re.search(r"@import", css, re.I):
        problems.append("CSS @import")
    for target in re.findall(r"url\(([^)]*)\)", css, re.I):
        if not target.strip(" '\"").startswith("#"):
            problems.append("external url() in CSS")

    if problems:
        unique = list(dict.fromkeys(problems))  # remove duplicates, keep order
        return False, "Unsafe content: " + "; ".join(unique[:4])
    return True, "No scripts, event handlers or external references"


# ------------------------------------------------------ run everything
def _as_dict(passed, reason):
    return {"passed": bool(passed), "reason": reason}


def validate_svg(svg_string):
    """Run all five checks. Returns {check_name: {"passed": bool, "reason": str}}."""
    checks = {}

    ok, reason = check_valid_xml(svg_string)
    checks["valid_xml"] = _as_dict(ok, reason)
    root = parse_svg(svg_string) if ok else None

    later_checks = [
        ("viewbox_bounds", check_viewbox_bounds),
        ("title_and_labels", check_title_and_labels),
        ("animation", check_animation),
        ("security", check_security),
    ]
    for name, function in later_checks:
        if root is None:
            checks[name] = _as_dict(False, "Skipped: the SVG could not be parsed")
            continue
        try:
            passed, reason = function(root)
        except Exception as error:  # a buggy check must never crash the pipeline
            passed, reason = False, f"Check crashed: {type(error).__name__}: {error}"
        checks[name] = _as_dict(passed, reason)

    return checks


def all_passed(checks):
    return all(check["passed"] for check in checks.values())


def failure_messages(checks):
    """List of 'name: reason' strings for failed checks. Phase 3 sends these to Gemini."""
    return [f"{name}: {c['reason']}" for name, c in checks.items() if not c["passed"]]

"""Generate diagrams and save them to data/results.json.

  python -m scripts.generate_all                 # all concepts that are missing or failed
  python -m scripts.generate_all --only 3        # just concept 3 (if missing/failed)
  python -m scripts.generate_all --only 3 --force  # regenerate 3 even if it passed
  python -m scripts.generate_all --force         # regenerate everything
"""
import argparse
import json
import logging
import os
import time

from app import config
from app.concepts import CONCEPTS
from app.services.orchestrator import GenerationError, generate_diagram


def load_results():
    """Existing results as {concept_id: result}. Empty if the file doesn't exist yet."""
    if config.RESULTS_FILE.exists():
        data = json.loads(config.RESULTS_FILE.read_text(encoding="utf-8"))
        return {item["concept_id"]: item for item in data}
    return {}


def save_results(by_id):
    """Write results in concept order. Writes to a temp file first, then swaps it in,
    so a crash mid-write can't leave a half-written results.json."""
    config.RESULTS_FILE.parent.mkdir(exist_ok=True)
    ordered = [by_id[cid] for cid in CONCEPTS if cid in by_id]
    temp = config.RESULTS_FILE.with_suffix(".json.tmp")
    # ensure_ascii=False keeps symbols like ² and ₂ readable in the file
    temp.write_text(json.dumps(ordered, indent=2, ensure_ascii=False), encoding="utf-8")
    os.replace(temp, config.RESULTS_FILE)


def main():
    parser = argparse.ArgumentParser(description="Generate science diagrams")
    parser.add_argument("--only", nargs="+", metavar="ID", help="concept ids, e.g. --only 2 4")
    parser.add_argument("--force", action="store_true", help="regenerate even if already passed")
    args = parser.parse_args()

    logging.basicConfig(level=logging.INFO, format="%(message)s")

    targets = args.only or list(CONCEPTS)
    unknown = [cid for cid in targets if cid not in CONCEPTS]
    if unknown:
        raise SystemExit(f"Unknown concept id(s): {unknown}. Valid ids: {list(CONCEPTS)}")

    results = load_results()
    errors = {}
    generated_any = False

    for cid in targets:
        existing = results.get(cid)
        if existing and existing["status"] == "passed" and not args.force:
            print(f"[{cid}] {CONCEPTS[cid]['name']}: already passed, skipping (use --force)")
            continue

        if generated_any:
            time.sleep(config.DELAY_BETWEEN_CONCEPTS)  # be gentle with the free tier
        generated_any = True

        try:
            results[cid] = generate_diagram(cid, profile="batch")
            save_results(results)   # save immediately
        except GenerationError as error:
            errors[cid] = error.message
            print(f"[{cid}] FAILED, keeping the old result if there is one: {error.message}")

    # ---- summary
    print("\n=== Summary ===")
    for cid in CONCEPTS:
        item = results.get(cid)
        if item:
            print(f"{cid}. {item['concept_name']:<22} {item['status']:<7} "
                  f"${item['cost_usd']:<10} retries={item['retries']['total']}")
        else:
            print(f"{cid}. {CONCEPTS[cid]['name']:<22} MISSING  {errors.get(cid, '')}")
    if len(results) < len(CONCEPTS):
        print("\nSome concepts are missing. Wait a minute, then run the script again.")


if __name__ == "__main__":
    main()

// ------------------------------------------------------------ small helpers
function el(tag, className, text) {
  const node = document.createElement(tag);
  if (className) node.className = className;
  if (text !== undefined) node.textContent = text;
  return node;
}

// SVG shown through <img> cannot run scripts. It needs xmlns to render.
function svgToDataUrl(svg) {
  if (!/<svg[^>]*\sxmlns\s*=/.test(svg)) {
    svg = svg.replace("<svg", '<svg xmlns="http://www.w3.org/2000/svg"');
  }
  return "data:image/svg+xml;charset=utf-8," + encodeURIComponent(svg);
}

function showError(message) {
  const banner = document.getElementById("error-banner");
  banner.textContent = message;
  banner.hidden = false;
}

function hideError() {
  document.getElementById("error-banner").hidden = true;
}

// ---------------------------------------------------- reviews (localStorage)
function loadReviews() {
  try {
    return JSON.parse(localStorage.getItem(REVIEWS_KEY)) || {};
  } catch (error) {
    return {};   // storage blocked or data damaged: start empty instead of crashing
  }
}

function saveReviews(reviews) {
  try {
    localStorage.setItem(REVIEWS_KEY, JSON.stringify(reviews));
  } catch (error) {
    showError("Could not save your review in this browser. Is storage blocked or full?");
  }
}

function updateReview(result, changes) {
  const reviews = loadReviews();
  const current = reviews[result.concept_id] || { status: null, note: "" };
  reviews[result.concept_id] = {
    ...current,
    ...changes,
    version: result.generated_at,          // which version of the diagram this was for
    updatedAt: new Date().toISOString(),
  };
  saveReviews(reviews);
}

function buildReviewCell(result) {
  const cell = el("td", "review-cell");
  const saved = loadReviews()[result.concept_id] || { status: null, note: "" };
  let current = saved.status;

  const statusText = el("div", "review-status");
  const approveBtn = el("button", "btn approve", "Approve");
  const rejectBtn = el("button", "btn reject", "Reject");
  const note = document.createElement("textarea");
  note.placeholder = "Reviewer note…";
  note.rows = 3;
  note.value = saved.note || "";
  note.setAttribute("aria-label", "Reviewer note for " + result.concept_name);

  const warning = el("div", "warn small",
    "This review was made for an earlier version of this diagram.");
  warning.hidden = !((saved.status || saved.note) && saved.version &&
                     saved.version !== result.generated_at);

  function paint() {
    approveBtn.classList.toggle("active", current === "approved");
    rejectBtn.classList.toggle("active", current === "rejected");
    approveBtn.setAttribute("aria-pressed", String(current === "approved"));
    rejectBtn.setAttribute("aria-pressed", String(current === "rejected"));
    statusText.textContent = current ? "Decision: " + current : "Not reviewed yet";
  }

  function choose(choice) {
    current = current === choice ? null : choice;   // clicking the active button clears it
    warning.hidden = true;
    updateReview(result, { status: current });
    paint();
  }

  approveBtn.addEventListener("click", () => choose("approved"));
  rejectBtn.addEventListener("click", () => choose("rejected"));
  note.addEventListener("input", () => {
    warning.hidden = true;
    updateReview(result, { note: note.value });
  });

  paint();
  const buttons = el("div", "btn-row");
  buttons.append(approveBtn, rejectBtn);
  cell.append(statusText, buttons, note, warning);
  return cell;
}

// ------------------------------------------------------------- table rows
function renderRow(result, isLive) {
  const row = el("tr");
  row.dataset.id = result.concept_id;

  // 1. Concept
  const conceptCell = el("td");
  conceptCell.append(
    el("strong", "", result.concept_name),
    el("div", "muted", result.subject),
    el("span", "badge " + (result.status === "passed" ? "pass" : "fail"),
       result.status === "passed" ? "All checks passed" : "Some checks failed"),
    el("div", "muted small", "Model: " + result.model)
  );
  if (isLive) {
    conceptCell.append(el("div", "badge live", "Live result (not saved on the server)"));
  }

  // 2. Diagram
  const diagramCell = el("td");
  if (result.svg) {
    const img = el("img", "diagram");
    img.src = svgToDataUrl(result.svg);
    img.alt = "Diagram: " + result.concept_name;
    diagramCell.append(img);
  } else {
    diagramCell.append(el("em", "muted", "No SVG was produced"));
  }

  // 3. Checks
  const checksCell = el("td");
  for (const [key, check] of Object.entries(result.checks)) {
    const line = el("div", "check");
    line.append(
      el("span", "badge " + (check.passed ? "pass" : "fail"),
         (check.passed ? "PASS " : "FAIL ") + (CHECK_LABELS[key] || key)),
      el("div", "muted small", check.reason)
    );
    checksCell.append(line);
  }

  // 4. Cost
  const costCell = el("td");
  costCell.append(
    el("strong", "", "$" + result.cost_usd.toFixed(6)),
    el("div", "muted small",
       `in ${result.tokens.input} / out ${result.tokens.output} / thinking ${result.tokens.thoughts} tokens`)
  );

  // 5. Retries
  const retriesCell = el("td");
  retriesCell.append(
    el("strong", "", String(result.retries.total)),
    el("div", "muted small",
       `API: ${result.retries.api}, validation: ${result.retries.validation}`)
  );

  row.append(conceptCell, diagramCell, checksCell, costCell, retriesCell,
             buildReviewCell(result));
  return row;
}

// Put a result in the table: replaces the concept's row, or adds it if new
function placeRow(result, isLive) {
  const newRow = renderRow(result, isLive);
  const old = rowsById[result.concept_id];
  if (old) {
    old.replaceWith(newRow);
  } else {
    document.getElementById("results-body").append(newRow);
  }
  rowsById[result.concept_id] = newRow;
}

async function generate() {

  const select = document.getElementById("concept-select");
  const button = document.getElementById("generate-btn");
  const status = document.getElementById("generate-status");
  const conceptId = select.value;

  generating = true;
  button.disabled = true;
  select.disabled = true;
  button.classList.add("loading");
  status.textContent = "Generating… this can take up to a minute.";
  hideError();
  if (rowsById[conceptId]) rowsById[conceptId].classList.add("busy");

  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), GENERATE_TIMEOUT_MS);

  try {
    let response;
    try {
      response = await fetch("/api/generate", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ concept_id: conceptId }),
        signal: controller.signal,
      });
    } catch (error) {
      throw new Error(
        error.name === "AbortError"
          ? "The request took too long and was cancelled. Please try again."
          : "Could not reach the server. If the site was idle it may still be waking up, so try again in a minute."
      );
    }

    let data = null;
    try { data = await response.json(); } catch (error) { /* not JSON */ }

    if (!response.ok) {
      throw new Error((data && data.error) ||
        `The server returned an error (status ${response.status}). Please try again.`);
    }
    if (!data || !data.checks) {
      throw new Error("The server sent an unreadable response.");
    }

    placeRow(data, true);
    status.textContent = data.status === "passed"
      ? `Done: ${data.concept_name} passed all checks.`
      : `Done: ${data.concept_name} failed some checks (see its row).`;
  } catch (error) {
    showError("Generation failed: " + error.message);
    status.textContent = "";
  } finally {
    clearTimeout(timer);
    generating = false;
    button.disabled = false;
    select.disabled = false;
    button.classList.remove("loading");
    if (rowsById[conceptId]) rowsById[conceptId].classList.remove("busy");
  }
}

async function init() {
  try {
    const response = await fetch("/api/results");
    if (!response.ok) throw new Error("Server returned " + response.status);
    const data = await response.json();
    if (!data.results.length) {
      showError("No saved results found. Run: python -m scripts.generate_all");
      return;
    }

    const select = document.getElementById("concept-select");
    data.results.forEach((result) => {
      placeRow(result, false);
      const option = el("option", "", `${result.concept_id}. ${result.concept_name}`);
      option.value = result.concept_id;
      select.append(option);
    });

    document.getElementById("results-table").hidden = false;
    select.disabled = false;
    const button = document.getElementById("generate-btn");
    button.disabled = false;
    button.addEventListener("click", generate);
  } catch (error) {
    showError("Could not load results: " + error.message);
  } finally {
    document.getElementById("loading").hidden = true;
  }
}

    def __init__(self, code):

def reply(text, inp=100, out=1000, thoughts=0):
    return {"text": text, "model": config.MODEL_NAME, "input_tokens": inp,
            "output_tokens": out, "thought_tokens": thoughts, "finish_reason": "STOP"}


class Fake:
    """Returns/raises the scripted items one by one and remembers the prompts."""
    def __init__(self, *script):
        self.script, self.prompts = list(script), []

    def __call__(self, prompt):
        self.prompts.append(prompt)
        item = self.script.pop(0)
        if isinstance(item, Exception):
            raise item
        return item


def run(fake, sleeps=None):
    sleeps = [] if sleeps is None else sleeps
    return generate_diagram("1", call_model=fake, sleep=sleeps.append)
